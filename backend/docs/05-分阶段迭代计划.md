## 三、阶段间自动化流转

### 3.1 编排器设计

**核心组件**: `backend/scripts/stage_runner.py` (Python)

```python
"""
Stage Runner: 阶段编排器
=======================
功能:
1. 读取 stage_config.yaml 当前阶段配置
2. 执行阶段任务 (调用 AI / 跑脚本)
3. 验证完成度 (跑测试 / 命令)
4. 自动生成阶段报告
5. 推进到下一阶段
"""
import yaml
import subprocess
from pathlib import Path
from datetime import datetime


class StageRunner:
    """阶段编排器 - 单人 + AI 协作"""
    
    def __init__(self, config_path="backend/docs/stage_config.yaml"):
        self.config = self._load_config(config_path)
        self.current_stage = self.config["current_stage"]
        self.log_path = Path("backend/docs/stage_logs")
        self.log_path.mkdir(exist_ok=True)
    
    def run(self):
        """执行当前阶段"""
        stage = self.config["stages"][self.current_stage]
        print(f"🚀 启动阶段: {stage['name']}")
        print(f"   目标: {stage['goal']}")
        print(f"   任务数: {len(stage['tasks'])}\n")
        
        # 1. 执行任务
        results = []
        for task in stage["tasks"]:
            print(f"▶ 执行任务: {task['id']} - {task['title']}")
            result = self._execute_task(task)
            results.append(result)
            self._print_task_status(result)
        
        # 2. 验证完成度
        completion = self._verify_completion(stage)
        print(f"\n📊 阶段完成度: {completion['percent']:.1f}%")
        
        # 3. 生成报告
        if completion["percent"] >= stage["exit_threshold"]:
            self._generate_report(stage, results, completion)
            self._advance_stage()
            print(f"✅ 阶段 {self.current_stage} 完成, 自动推进到 {self._next_stage()}\n")
        else:
            print(f"⚠️ 完成度 {completion['percent']:.1f}% < 阈值 {stage['exit_threshold']}%")
            print(f"   剩余任务: {completion['pending_tasks']}")
            self._save_report(stage, results, completion, status="INCOMPLETE")
    
    def _execute_task(self, task):
        """执行单个任务 (调用 AI / 跑命令)"""
        log_file = self.log_path / f"{task['id']}.log"
        try:
            if task.get("ai_assisted"):
                # AI 协助: 调用 AI 生成代码/命令
                result = self._ai_execute(task)
            else:
                # 人工: 等待用户标记完成 (--mark-complete)
                result = self._wait_human(task)
            
            return {
                "task_id": task["id"],
                "status": "completed" if result["ok"] else "failed",
                "log": str(log_file),
                "output": result.get("output", ""),
                "timestamp": datetime.now().isoformat(),
            }
        except Exception as e:
            return {
                "task_id": task["id"],
                "status": "failed",
                "error": str(e),
                "timestamp": datetime.now().isoformat(),
            }
    
    def _verify_completion(self, stage):
        """验证阶段完成度"""
        total = len(stage["tasks"])
        completed = sum(1 for t in stage["tasks"] if self._is_completed(t))
        return {
            "percent": completed / total * 100,
            "completed": completed,
            "total": total,
            "pending_tasks": [t["id"] for t in stage["tasks"] if not self._is_completed(t)],
        }
    
    def _is_completed(self, task):
        """判断任务是否完成 (基于完成标记文件)"""
        marker = Path(f"backend/docs/.task_done/{task['id']}.done")
        return marker.exists()
    
    def _generate_report(self, stage, results, completion):
        """生成阶段报告 S{N}_REPORT.md"""
        template = self._load_template("stage_report.md")
        report = template.format(
            stage_id=stage["id"],
            stage_name=stage["name"],
            goal=stage["goal"],
            start_date=results[0]["timestamp"] if results else "",
            end_date=datetime.now().isoformat(),
            completion=completion,
            tasks=stage["tasks"],
            results=results,
        )
        report_path = Path(f"backend/docs/S{stage['id'][1:]}_REPORT.md")
        report_path.write_text(report, encoding="utf-8")
        # 自动 git add + commit
        subprocess.run(["git", "add", str(report_path)])
        subprocess.run(["git", "commit", "-m", f"docs(stage): S{stage['id'][1:]} {stage['name']} 阶段完成报告"])
    
    def _advance_stage(self):
        """推进到下一阶段"""
        # 更新 stage_config.yaml 的 current_stage
        next_stage = self._next_stage()
        self.config["current_stage"] = next_stage
        self._save_config()
    
    def _next_stage(self):
        """获取下一阶段 ID"""
        order = list(self.config["stages"].keys())
        idx = order.index(self.current_stage)
        return order[idx + 1] if idx + 1 < len(order) else "DONE"


if __name__ == "__main__":
    import sys
    stage = sys.argv[1] if len(sys.argv) > 1 else None
    runner = StageRunner()
    if stage:
        runner.current_stage = stage
    runner.run()
```

- **可恢复**: 任何阶段可暂停/回滚, 不会卡死
